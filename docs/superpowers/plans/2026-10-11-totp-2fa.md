# Mandatory TOTP 2FA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Require authenticator-app TOTP for every user before issuing a full session, with backup codes, merchant 30-day trusted devices, and SUPERADMIN reset.

**Architecture:** Password success mints a short-lived `mfa_pending` JWT (purpose claim only). Full access + refresh tokens are issued only after setup confirm or verify (or merchant trusted-device match). Pending TOTP secrets live in Redis until confirm; enabled secrets are AES-GCM encrypted via existing `encryptVault` / `SESSION_VAULT_KEY`.

**Tech Stack:** Express + Prisma + Vitest + jose + otplib + Redis (apis); Next.js App Router BFF + `proxy.ts` + React Hook Form (portal).

**Spec:** `docs/superpowers/specs/2026-10-11-totp-2fa-design.md`

## Global Constraints

- Dual repos: commit API work in `affiliate-tool-apis`, portal work in `affiliate-tool-portal`
- TOTP required for everyone — no self-disable, no grace period
- Never issue full access/refresh until MFA complete (including `register`)
- Pending token purpose exactly `mfa_pending`; normal `authenticate` must reject it
- Encrypt TOTP secrets with existing `encryptVault` / `decryptVault` (`SESSION_VAULT_KEY`) — no separate `TOTP_ENCRYPTION_KEY`
- Backup codes: 10, hashed with `sha256`, single-use, returned once
- Trusted device: merchants only, 30 days; staff always challenged
- Admin reset: `SUPERADMIN` only + audit + revoke refresh + devices
- Pending setup secret in Redis only (key `mfa-setup:{userId}`), TTL = pending token TTL
- Rate-limit verify/setup: 5/min per IP (failClosed)
- Do not log secrets, backup codes, or otpauth URLs

```ts
export async function signMfaPendingToken(userId: string): Promise<string> {
  return new SignJWT({ purpose: "mfa_pending" })
    .setProtectedHeader({ alg: algorithm() })
    .setSubject(userId)
    .setIssuedAt()
    .setExpirationTime(`${env.MFA_PENDING_TTL_SEC}s`)
    .sign(await getSigningKey());
}

export async function verifyMfaPendingToken(
  token: string,
): Promise<{ sub: string }> {
  const { payload } = await jwtVerify(token, await getVerifyKey(), {
    algorithms: [algorithm()],
  });
  if (payload.purpose !== "mfa_pending" || !payload.sub) {
    throw new Error("Invalid mfa pending token");
  }
  return { sub: String(payload.sub) };
}
```

Update `verifyAccessToken` / `claimsFromPayload` path: if `payload.purpose` is set, throw (do not treat as access token).

- [ ] **Step 5: Implement `src/lib/totp.ts`**

```ts
import { authenticator } from "otplib";
import { randomBytes } from "node:crypto";
import { sha256 } from "./crypto.js";

authenticator.options = { step: 30, window: 1, digits: 6 };

const ISSUER = "Tiksly";

export function generateTotpSecret(): string {
  return authenticator.generateSecret();
}

export function buildOtpauthUrl(email: string, secret: string): string {
  return authenticator.keyuri(email, ISSUER, secret);
}

export function verifyTotpCode(secret: string, code: string): boolean {
  const normalized = code.replace(/\s/g, "");
  return authenticator.check(normalized, secret);
}

export function generateBackupCodes(): { raw: string[]; hashes: string[] } {
  const raw = Array.from({ length: 10 }, () =>
    randomBytes(5).toString("hex"), // 10 hex chars
  );
  return { raw, hashes: raw.map(hashBackupCode) };
}

export function hashBackupCode(raw: string): string {
  return sha256(raw.trim().toLowerCase());
}
```

export const authenticateMfaPending: RequestHandler = async (req, _res, next) => {
  try {
    const header = req.headers.authorization;
    if (!header?.startsWith("Bearer ")) {
      throw new AppError("UNAUTHORIZED", "Missing MFA pending token", 401);

```ts
export async function startTotpSetup(userId: string) {
  const user = await prisma.user.findUniqueOrThrow({ where: { id: userId } });
  if (user.totpEnabled) {
    throw new AppError("CONFLICT", "Two-factor authentication already enabled", 409);
  }
  const secret = generateTotpSecret();
  await setPendingTotpSecret(userId, encryptVault(secret), env.MFA_PENDING_TTL_SEC);
  // Actually store plaintext secret encrypted once: setPendingTotpSecret should encrypt internally OR accept plaintext and encrypt inside store — pick one: store encrypts plaintext.
  return { otpauthUrl: buildOtpauthUrl(user.email, secret) };
}

export async function confirmTotpSetup(userId: string, code: string) {
  const user = await prisma.user.findUniqueOrThrow({ where: { id: userId } });
  if (user.totpEnabled) throw new AppError("CONFLICT", "...", 409);
  const secret = await getPendingTotpSecret(userId); // decrypted plaintext
  if (!secret || !verifyTotpCode(secret, code)) {
    throw new AppError("UNAUTHORIZED", "Invalid verification code", 401);
  }
  const { raw, hashes } = generateBackupCodes();
  await prisma.$transaction(async (tx) => {
    await tx.user.update({
      where: { id: userId },
      data: {
        totpEnabled: true,
        totpSecretEncrypted: encryptVault(secret),
        totpVerifiedAt: new Date(),
      },
    });
    await tx.twoFactorBackupCode.createMany({
      data: hashes.map((codeHash) => ({ userId, codeHash })),
    });
  });
  await deletePendingTotpSecret(userId);
  const { claims, platformMembership } = await resolveAccessClaims(userId);
  // export resolveAccessClaims or duplicate thin wrapper — prefer exporting from auth.service or moving claims helper to shared
  const session = await issueSession(claims);
  void writeAuditLog({ action: "auth.2fa.enabled", ... });
  return {
    status: "authenticated" as const,
    backupCodes: raw,
    user: { id: user.id, email: user.email, name: user.name },
    organizationId: claims.orgId,
    platformMembership,
    redirectTo: redirectFor(claims.platformRole, claims.hasProductAccess),
    ...session,
  };
}
```

**Note:** Export `resolveAccessClaims` and `redirectFor` from `auth.service.ts` (or move to `auth-claims.ts`) so `two-factor.service.ts` can use them without circular imports. Prefer small `auth-claims.ts` if circular risk appears.

```ts
export async function verifyTotpLogin(
  userId: string,
  code: string,
  opts: { rememberDevice?: boolean; userAgent?: string },
) {
  const user = await prisma.user.findUniqueOrThrow({ where: { id: userId } });
  if (!user.totpEnabled || !user.totpSecretEncrypted) {
    throw new AppError("FORBIDDEN", "Two-factor authentication is not enabled", 403);
  }
  const secret = decryptVault(user.totpSecretEncrypted);
  const totpOk = verifyTotpCode(secret, code);
  let backupUsed = false;
  if (!totpOk) {
    backupUsed = await consumeBackupCode(userId, code);
    if (!backupUsed) {
      throw new AppError("UNAUTHORIZED", "Invalid verification code", 401);
    }
  }

  const { claims, platformMembership } = await resolveAccessClaims(userId);
  const session = await issueSession(claims);

  let trustedDeviceRaw: string | undefined;
  if (opts.rememberDevice && !claims.platformRole) {
    trustedDeviceRaw = randomBytes(32).toString("base64url");
    await prisma.trustedDevice.create({
      data: {
        userId,
        tokenHash: sha256(trustedDeviceRaw),
        label: opts.userAgent?.slice(0, 120) ?? null,
        expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000),
      },
    });
  }

  void writeAuditLog({
    action: "auth.2fa.verified",
    meta: { backupUsed, rememberDevice: Boolean(trustedDeviceRaw) },
    ...
  });

  return { status: "authenticated" as const, trustedDeviceRaw, ...sessionFields };
}
```

`consumeBackupCode`: find unused row where `codeHash === hashBackupCode(code)`, set `usedAt`.
